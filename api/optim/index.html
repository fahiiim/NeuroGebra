    class SGD(Optimizer):
    """
    Stochastic Gradient Descent optimizer with optional momentum and Nesterov.

    Args:
        lr: Learning rate.
        momentum: Momentum factor (default: 0.0).
        nesterov: Enables Nesterov momentum (default: False).
        weight_decay: Weight decay (L2 penalty) factor (default: 0.0).
    """

    def __init__(
        self,
        lr: float = 0.01,
        momentum: float = 0.0,
        nesterov: bool = False,
        weight_decay: float = 0.0,
    ):
        super().__init__(lr=lr)
        self.momentum = momentum
        self.nesterov = nesterov
        self.weight_decay = weight_decay
        self.velocities: list[np.ndarray | None] = []

    def step(self, params: list[Any]) -> None:
        self.t += 1
        if not self.velocities:
            self.velocities = [
                np.zeros_like(p.data) if hasattr(p, "data") else None for p in params
            ]

        for i, p in enumerate(params):
            if not hasattr(p, "grad") or p.grad is None:
                continue

            grad = np.asarray(p.grad, dtype=np.float64)
            if self.weight_decay != 0:
                grad = grad + self.weight_decay * p.data

            if self.momentum != 0:
                if self.velocities[i] is None:
                    self.velocities[i] = np.zeros_like(p.data)
                self.velocities[i] = self.momentum * self.velocities[i] + grad
                if self.nesterov:
                    step_val = self.momentum * self.velocities[i] + grad
                else:
                    step_val = self.velocities[i]
                p.data -= self.learning_rate * step_val
            else:
                p.data -= self.learning_rate * grad

    class Adam(Optimizer):
    """
    Adam (Adaptive Moment Estimation) optimizer.

    Args:
        lr: Learning rate (default: 0.001).
        betas: Coefficients used for computing running averages (default: (0.9, 0.999)).
        eps: Term added to the denominator for numerical stability (default: 1e-8).
        weight_decay: Weight decay (L2 penalty) factor (default: 0.0).
    """

    def __init__(
        self,
        lr: float = 0.001,
        betas: tuple[float, float] = (0.9, 0.999),
        eps: float = 1e-8,
        weight_decay: float = 0.0,
    ):
        super().__init__(lr=lr)
        self.beta1, self.beta2 = betas
        self.eps = eps
        self.weight_decay = weight_decay
        self.m: list[np.ndarray | None] = []
        self.v: list[np.ndarray | None] = []

    def step(self, params: list[Any]) -> None:
        self.t += 1
        if not self.m:
            self.m = [
                np.zeros_like(p.data) if hasattr(p, "data") else None for p in params
            ]
            self.v = [
                np.zeros_like(p.data) if hasattr(p, "data") else None for p in params
            ]

        lr = self.learning_rate
        b1, b2 = self.beta1, self.beta2
        eps = self.eps
        bias_correction1 = 1.0 - b1**self.t
        bias_correction2 = 1.0 - b2**self.t

        for i, p in enumerate(params):
            if not hasattr(p, "grad") or p.grad is None:
                continue

            grad = np.asarray(p.grad, dtype=np.float64)
            if self.weight_decay != 0:
                grad = grad + self.weight_decay * p.data

            if self.m[i] is None:
                self.m[i] = np.zeros_like(p.data)
                self.v[i] = np.zeros_like(p.data)

            self.m[i] = b1 * self.m[i] + (1 - b1) * grad
            self.v[i] = b2 * self.v[i] + (1 - b2) * (grad**2)

            m_hat = self.m[i] / bias_correction1
            v_hat = self.v[i] / bias_correction2

            p.data -= lr * m_hat / (np.sqrt(v_hat) + eps)

    class RMSprop(Optimizer):
    """
    RMSprop optimizer.

    Args:
        lr: Learning rate (default: 0.01).
        alpha: Smoothing constant (default: 0.99).
        eps: Term added to the denominator for numerical stability (default: 1e-8).
        weight_decay: Weight decay (L2 penalty) factor (default: 0.0).
    """

    def __init__(
        self,
        lr: float = 0.01,
        alpha: float = 0.99,
        eps: float = 1e-8,
        weight_decay: float = 0.0,
    ):
        super().__init__(lr=lr)
        self.alpha = alpha
        self.eps = eps
        self.weight_decay = weight_decay
        self.square_avgs: list[np.ndarray | None] = []

    def step(self, params: list[Any]) -> None:
        self.t += 1
        if not self.square_avgs:
            self.square_avgs = [
                np.zeros_like(p.data) if hasattr(p, "data") else None for p in params
            ]

        for i, p in enumerate(params):
            if not hasattr(p, "grad") or p.grad is None:
                continue

            grad = np.asarray(p.grad, dtype=np.float64)
            if self.weight_decay != 0:
                grad = grad + self.weight_decay * p.data

            if self.square_avgs[i] is None:
                self.square_avgs[i] = np.zeros_like(p.data)

            self.square_avgs[i] = self.alpha * self.square_avgs[i] + (
                1.0 - self.alpha
            ) * (grad**2)
            avg = np.sqrt(self.square_avgs[i]) + self.eps
            p.data -= self.learning_rate * grad / avg

    class StepLR(LRScheduler):
    """
    Decays the learning rate by gamma every step_size epochs.

    Formula:
        lr = base_lr * (gamma ** (epoch // step_size))
    """

    def __init__(
        self,
        target: Any,
        step_size: int,
        gamma: float = 0.1,
        last_epoch: int = -1,
    ):
        if step_size <= 0:
            raise ValueError(f"step_size must be positive, got {step_size}")
        self.step_size = step_size
        self.gamma = gamma
        super().__init__(target, last_epoch)

    def get_lr(self) -> float:
        if self.last_epoch == 0:
            return self.base_lr
        return self.base_lr * (self.gamma ** (self.last_epoch // self.step_size))

    class CosineAnnealingLR(LRScheduler):
    """
    Cosine annealing learning rate schedule.

    Formula:
        lr = eta_min + 0.5 * (base_lr - eta_min) * (1 + cos(pi * epoch / T_max))
    """

    def __init__(
        self,
        target: Any,
        T_max: int,
        eta_min: float = 0.0,
        last_epoch: int = -1,
    ):
        if T_max <= 0:
            raise ValueError(f"T_max must be positive, got {T_max}")
        self.T_max = T_max
        self.eta_min = eta_min
        super().__init__(target, last_epoch)

    def get_lr(self) -> float:
        if self.last_epoch == 0:
            return self.base_lr
        fraction = min(1.0, self.last_epoch / self.T_max)
        return self.eta_min + 0.5 * (self.base_lr - self.eta_min) * (
            1.0 + math.cos(math.pi * fraction)
        )

    class ExponentialLR(LRScheduler):
    """
    Decays the learning rate by gamma every epoch.

    Formula:
        lr = base_lr * (gamma ** epoch)
    """

    def __init__(
        self,
        target: Any,
        gamma: float = 0.95,
        last_epoch: int = -1,
    ):
        if gamma <= 0:
            raise ValueError(f"gamma must be positive, got {gamma}")
        self.gamma = gamma
        super().__init__(target, last_epoch)

    def get_lr(self) -> float:
        return self.base_lr * (self.gamma**self.last_epoch)

    class ReduceLROnPlateau:
    """
    Reduce learning rate when a metric has stopped improving.

    Args:
        target: Model or Optimizer having learning_rate or lr.
        mode: 'min' or 'max'.
        factor: Factor by which the learning rate will be reduced. lr = lr * factor.
        patience: Number of epochs with no improvement after which lr is reduced.
        min_lr: Lower bound on the learning rate.
    """

    def __init__(
        self,
        target: Any,
        mode: str = "min",
        factor: float = 0.5,
        patience: int = 5,
        min_lr: float = 1e-6,
    ):
        self.target = target
        if hasattr(target, "learning_rate"):
            self.current_lr = float(target.learning_rate)
        elif hasattr(target, "lr"):
            self.current_lr = float(target.lr)
        else:
            self.current_lr = 0.001

        self.mode = mode.lower()
        self.factor = factor
        self.patience = patience
        self.min_lr = min_lr
        self.best = float("inf") if self.mode == "min" else float("-inf")
        self.num_bad_epochs = 0

    def step(self, metric: float) -> float:
        """
        Update the learning rate based on current epoch metric.

        Args:
            metric: Loss or performance metric value to evaluate.

        Returns:
            The updated learning rate.
        """
        is_better = (
            (metric < self.best) if self.mode == "min" else (metric > self.best)
        )
        if is_better:
            self.best = metric
            self.num_bad_epochs = 0
        else:
            self.num_bad_epochs += 1

        if self.num_bad_epochs > self.patience:
            self.current_lr = max(self.min_lr, self.current_lr * self.factor)
            if hasattr(self.target, "learning_rate"):
                self.target.learning_rate = self.current_lr
            elif hasattr(self.target, "lr"):
                self.target.lr = self.current_lr
            self.num_bad_epochs = 0

        return self.current_lr
