    Hardware Acceleration Subsystem¶

    The neurogebra.acceleration module compiles supported SymPy expressions into portable native C libraries or NVIDIA CUDA kernels through NVRTC. See the backend guide for precision, toolchain and execution boundaries.

    def compile_c(
    expr: str | sp.Expr,
    variables: list[str | sp.Symbol],
    opt_level: str = "-O3",
) -> CJITFunction:
    """
    Compile a symbolic expression into a high-performance native C shared library.

    Args:
        expr: SymPy expression or mathematical string
        variables: List of variable symbols or names
        opt_level: Compiler optimization level (default -O3)

    Returns:
        CJITFunction ready for native evaluation.
    """
    compiler = find_c_compiler()
    if compiler is None:
        raise RuntimeError(
            "No C/C++ compiler found (clang++, g++, gcc). Please install a compiler."
        )
    if opt_level not in {"-O0", "-O1", "-O2", "-O3", "-Os"}:
        raise ValueError("opt_level must be -O0, -O1, -O2, -O3 or -Os")

    if isinstance(expr, str):
        sym_expr = sp.sympify(expr)
    else:
        sym_expr = expr

    var_symbols = [sp.Symbol(v) if isinstance(v, str) else v for v in variables]
    var_names = [v.name for v in var_symbols]
    if len(set(var_names)) != len(var_names):
        raise ValueError("Compilation variables must be unique")
    if sym_expr.free_symbols - set(var_symbols):
        raise ValueError("Every free symbol must be included in variables")

    # Mangle symbols to prevent collisions with C keywords, loop indices, or names like 'i', 'n'
    sym_map = {v: sp.Symbol(f"__ng_v_{k}") for k, v in enumerate(var_symbols)}
    mangled_expr = sym_expr.subs(sym_map)
    c_math = ccode(mangled_expr)

    # Build C source with robust symbol mangling
    source_lines = [
        "#include <math.h>",
        "#include <stdlib.h>",
        "",
        "#if defined(_WIN32) || defined(__WIN32__)",
        "#define EXPORT __declspec(dllexport)",
        "#else",
        '#define EXPORT __attribute__((visibility("default")))',
        "#endif",
        "",
        "#ifdef __cplusplus",
        'extern "C" {',
        "#endif",
        "",
        "EXPORT void evaluate_expr(",
        "    const double** in_ptrs,",
        "    const int* in_strides,",
        "    double* out,",
        "    int __ng_n",
        ") {",
    ]

    # Declare pointer aliases with mangled names
    for k in range(len(var_names)):
        source_lines.append(f"    const double* __ng_ptr_{k} = in_ptrs[{k}];")
        source_lines.append(f"    int __ng_stride_{k} = in_strides[{k}];")

    source_lines.append("")
    source_lines.append("    #pragma clang loop vectorize(enable)")
    source_lines.append("    #pragma omp simd")
    source_lines.append("    for (int __ng_idx = 0; __ng_idx < __ng_n; ++__ng_idx) {")

    for k in range(len(var_names)):
        source_lines.append(
            f"        double __ng_v_{k} = __ng_ptr_{k}[__ng_idx * __ng_stride_{k}];"
        )

    source_lines.append(f"        out[__ng_idx] = {c_math};")
    source_lines.append("    }")
    source_lines.append("}")
    source_lines.append("")
    source_lines.append("#ifdef __cplusplus")
    source_lines.append("}")
    source_lines.append("#endif")
    source_lines.append("")

    c_source = "\n".join(source_lines)

    # Compute content hash for disk caching
    cache_dir = _get_cache_dir()
    machine = platform.machine().lower()
    compiler_path = Path(compiler).resolve()
    compiler_stat = compiler_path.stat()
    compiler_key = (
        f"{compiler_path}:{compiler_stat.st_size}:{compiler_stat.st_mtime_ns}"
    )
    src_hash = hashlib.sha256(
        (
            c_source + opt_level + os.name + machine + compiler_key + "portable-v3"
        ).encode("utf-8")
    ).hexdigest()[:16]

    if sys.platform == "win32":
        suffix = ".dll"
    elif sys.platform == "darwin":
        suffix = ".dylib"
    else:
        suffix = ".so"

    dll_path = cache_dir / f"neuro_jit_{src_hash}{suffix}"
    src_path = cache_dir / f"neuro_jit_{src_hash}.c"

    if not dll_path.exists():
        src_path.write_text(c_source, encoding="utf-8")

        shared_flag = "-dynamiclib" if sys.platform == "darwin" else "-shared"

        base_cmd = [compiler, opt_level, shared_flag]
        if sys.platform != "win32":
            base_cmd.append("-fPIC")

        # A compiler accepting -mavx2 says nothing about the host CPU.
        # Use its portable target instead of risking an illegal instruction.
        cmd_candidates = [base_cmd + [str(src_path), "-o", str(dll_path)]]

        last_error = ""
        success = False
        for compile_cmd in cmd_candidates:
            try:
                proc = subprocess.run(
                    compile_cmd, capture_output=True, text=True, timeout=30
                )
                if proc.returncode == 0:
                    success = True
                    break
                else:
                    last_error = proc.stderr
            except Exception as e:
                last_error = str(e)

        if not success:
            raise RuntimeError(
                f"Failed to compile C-JIT kernel: {last_error}\nSource was:\n{c_source}"
            )

    return CJITFunction(dll_path, "evaluate_expr", var_names)

    def is_c_available() -> bool:
    """Check if a C/C++ compiler is available and functional for JIT compilation."""
    global _C_AVAILABLE_CACHE
    if _C_AVAILABLE_CACHE is not None:
        return _C_AVAILABLE_CACHE
    compiler = find_c_compiler()
    if compiler is None:
        _C_AVAILABLE_CACHE = False
        return False
    try:
        fn = compile_c("x + 1.0", ["x"])
        res = fn(np.array([1.0, 2.0], dtype=np.float64))
        _C_AVAILABLE_CACHE = bool(np.allclose(res, [2.0, 3.0]))
    except Exception:
        _C_AVAILABLE_CACHE = False
    return _C_AVAILABLE_CACHE

    def find_c_compiler() -> str | None:
    """Find an available C/C++ compiler on the system."""
    for compiler in ["clang++", "g++", "gcc", "clang"]:
        path = shutil.which(compiler)
        if path:
            return path
    return None

    class CJITFunction:
    """
    Callable wrapper for a native C compiled expression.
    """

    def __init__(
        self,
        dll_path: Path,
        func_name: str,
        var_names: list[str],
        dll_handle: ctypes.CDLL | None = None,
    ):
        self.dll_path = dll_path
        self.func_name = func_name
        self.var_names = var_names
        self.num_vars = len(var_names)
        self._dll = dll_handle or ctypes.CDLL(str(dll_path))
        self._c_func = getattr(self._dll, func_name)

        # C signature: void evaluate(const double** in_ptrs, const int* in_strides, double* out, int n)
        self._c_func.argtypes = [
            ctypes.POINTER(ctypes.c_void_p),
            ctypes.POINTER(ctypes.c_int),
            ctypes.POINTER(ctypes.c_double),
            ctypes.c_int,
        ]
        self._c_func.restype = None

    def __call__(self, *args: Any) -> float | np.ndarray:
        if len(args) != self.num_vars:
            raise ValueError(
                f"Expected {self.num_vars} arguments for variables {self.var_names}, got {len(args)}"
            )

        # Convert args to numpy arrays or scalars
        np_args: list[np.ndarray] = []
        is_all_scalar = True

        for a in args:
            if isinstance(a, np.ndarray):
                arr = np.asanyarray(a, dtype=np.float64)
                if arr.ndim > 0:
                    is_all_scalar = False
                np_args.append(arr)
            elif isinstance(a, (int, float, np.floating, np.integer)):
                np_args.append(np.array(float(a), dtype=np.float64))
            else:
                arr = np.asanyarray(a, dtype=np.float64)
                if arr.ndim > 0:
                    is_all_scalar = False
                np_args.append(arr)

        if is_all_scalar:
            # All scalar inputs
            in_ptrs_arr = (ctypes.c_void_p * self.num_vars)()
            in_strides_arr = (ctypes.c_int * self.num_vars)()
            for i, arr in enumerate(np_args):
                in_ptrs_arr[i] = arr.ctypes.data
                in_strides_arr[i] = 0

            out_val = np.empty(1, dtype=np.float64)
            self._c_func(
                in_ptrs_arr,
                in_strides_arr,
                out_val.ctypes.data_as(ctypes.POINTER(ctypes.c_double)),
                1,
            )
            return float(out_val[0])

        # NumPy standard broadcasting across all inputs
        broadcasted = np.broadcast_arrays(*np_args)
        broadcast_shape = broadcasted[0].shape
        total_len = broadcasted[0].size

        # Handle empty inputs gracefully without native kernel launch
        if total_len == 0:
            return np.empty(broadcast_shape, dtype=np.float64)

        b_contiguous = [np.ascontiguousarray(b, dtype=np.float64) for b in broadcasted]
        out = np.empty(total_len, dtype=np.float64)

        in_ptrs_arr = (ctypes.c_void_p * self.num_vars)()
        in_strides_arr = (ctypes.c_int * self.num_vars)()

        for i, arr in enumerate(b_contiguous):
            in_ptrs_arr[i] = arr.ctypes.data
            in_strides_arr[i] = 1

        self._c_func(
            in_ptrs_arr,
            in_strides_arr,
            out.ctypes.data_as(ctypes.POINTER(ctypes.c_double)),
            total_len,
        )

        return out.reshape(broadcast_shape)

    def compile_cuda(
    expr: str | sp.Expr,
    variables: list[str | sp.Symbol],
    arch: str | None = None,
) -> CUDAJITFunction:
    """
    Compile a symbolic mathematical expression to an optimized CUDA kernel via NVRTC.

    Args:
        expr: SymPy expression or mathematical string
        variables: List of variable symbols or names
        arch: Target GPU architecture (defaults to active GPU capability or compute_80)

    Returns:
        CUDAJITFunction ready to run on the GPU.
    """
    if not is_cuda_available():
        raise RuntimeError(
            "CUDA is not available on this machine. Ensure an NVIDIA GPU and driver are installed."
        )

    if arch is None:
        major, minor = get_cuda_compute_capability()
        if major >= 9:
            arch = "compute_90"
        elif major > 0:
            arch = f"compute_{major}{minor}"
        else:
            arch = "compute_80"

    if isinstance(expr, str):
        sym_expr = sp.sympify(expr)
    else:
        sym_expr = expr

    var_symbols = [sp.Symbol(v) if isinstance(v, str) else v for v in variables]
    var_names = [v.name for v in var_symbols]

    # Mangle symbols to prevent collisions in CUDA kernel
    sym_map = {v: sp.Symbol(f"__ng_v_{k}") for k, v in enumerate(var_symbols)}
    mangled_expr = sym_expr.subs(sym_map)
    c_math = ccode(mangled_expr)

    # Build CUDA C source
    params_sig = []
    for k in range(len(var_names)):
        params_sig.append(f"const float* __ng_ptr_{k}, int __ng_stride_{k}")
    params_sig.append("float* out, int __ng_n")
    sig_str = ", ".join(params_sig)

    cu_lines = [
        f'extern "C" __global__ void evaluate_kernel({sig_str}) {{',
        "    int __ng_idx = blockDim.x * blockIdx.x + threadIdx.x;",
        "    if (__ng_idx < __ng_n) {",
    ]

    for k in range(len(var_names)):
        cu_lines.append(
            f"        float __ng_v_{k} = __ng_ptr_{k}[__ng_idx * __ng_stride_{k}];"
        )

    cu_lines.append(f"        out[__ng_idx] = {c_math};")
    cu_lines.append("    }")
    cu_lines.append("}")
    cu_source = "\n".join(cu_lines)

    # Compile with NVRTC
    err, prog = nvrtc.nvrtcCreateProgram(
        cu_source.encode("utf-8"), b"expr_kernel.cu", 0, [], []
    )
    if err.value != 0:
        raise RuntimeError(f"nvrtcCreateProgram failed: {err}")

    try:
        opts = [f"--gpu-architecture={arch}".encode()]
        (comp_err,) = nvrtc.nvrtcCompileProgram(prog, len(opts), opts)
        if comp_err.value != 0:
            err, log_size = nvrtc.nvrtcGetProgramLogSize(prog)
            log = b" " * log_size
            nvrtc.nvrtcGetProgramLog(prog, log)
            raise RuntimeError(f"NVRTC compilation failed:\n{log.decode('utf-8')}")

        err, size = nvrtc.nvrtcGetPTXSize(prog)
        ptx = bytearray(size)
        (err,) = nvrtc.nvrtcGetPTX(prog, ptx)
    finally:
        nvrtc.nvrtcDestroyProgram(prog)

    return CUDAJITFunction(bytes(ptx), "evaluate_kernel", var_names)

    Callable wrapper for an NVRTC-compiled CUDA GPU kernel.
    """

    def __init__(
        self,
        ptx_bytes: bytes,
        kernel_name: str,
        var_names: list[str],
    ):
        self.ptx_bytes = ptx_bytes
        self.kernel_name = kernel_name
        self.var_names = var_names
        self.num_vars = len(var_names)

        # Initialize CUDA Context
        assert driver.cuInit(0)[0].value == 0
        err, self.dev = driver.cuDeviceGet(0)
        err, self.ctx = driver.cuCtxCreate(None, 0, self.dev)
        if err.value != 0:
            raise RuntimeError(f"CUDA context creation failed: {err}")

        err, self.module = driver.cuModuleLoadData(self.ptx_bytes)
        if err.value != 0:
            raise RuntimeError(f"CUDA module load failed: {err}")

        err, self.kernel = driver.cuModuleGetFunction(
            self.module, kernel_name.encode("utf-8")
        )
        if err.value != 0:
            raise RuntimeError(f"CUDA get function '{kernel_name}' failed: {err}")

    def __call__(self, *args: Any) -> float | np.ndarray:
        if len(args) != self.num_vars:
            raise ValueError(
                f"Expected {self.num_vars} arguments for variables {self.var_names}, got {len(args)}"
            )

        # Convert args to numpy float32 arrays (standard GPU precision)
        np_args: list[np.ndarray] = []
        is_all_scalar = True

        for a in args:
            if isinstance(a, np.ndarray):
                arr = np.asanyarray(a, dtype=np.float32)
                if arr.ndim > 0:
                    is_all_scalar = False
                np_args.append(arr)
            elif isinstance(a, (int, float, np.floating, np.integer)):
                np_args.append(np.array(float(a), dtype=np.float32))
            else:
                arr = np.asanyarray(a, dtype=np.float32)
                if arr.ndim > 0:
                    is_all_scalar = False
                np_args.append(arr)

        # Standard NumPy broadcasting
        broadcasted = np.broadcast_arrays(*np_args)
        broadcast_shape = broadcasted[0].shape
        total_len = broadcasted[0].size

        # Handle empty inputs gracefully without GPU kernel launch
        if total_len == 0:
            return np.empty(broadcast_shape, dtype=np.float32)

        b_contiguous = [np.ascontiguousarray(b, dtype=np.float32) for b in broadcasted]

        # Push context
        driver.cuCtxPushCurrent(self.ctx)
        d_inputs = []
        d_out = None
        try:
            for arr in b_contiguous:
                err, d_ptr = driver.cuMemAlloc(arr.nbytes)
                if err.value != 0:
                    raise RuntimeError(f"cuMemAlloc failed: {err}")
                d_inputs.append(d_ptr)
                err = driver.cuMemcpyHtoD(d_ptr, arr.ctypes.data, arr.nbytes)
                if hasattr(err, "value") and err.value != 0:
                    raise RuntimeError(f"cuMemcpyHtoD failed: {err}")

            out_bytes = total_len * 4  # float32 = 4 bytes
            err, d_out = driver.cuMemAlloc(out_bytes)
            if err.value != 0:
                raise RuntimeError(f"cuMemAlloc failed: {err}")

            # Build argument pointers
            ctypes_ptrs = []
            c_vars = []

            for i, _arr in enumerate(b_contiguous):
                d_val = ctypes.c_uint64(int(d_inputs[i]))
                s_val = ctypes.c_int(1)
                c_vars.extend([d_val, s_val])
                ctypes_ptrs.extend(
                    [
                        ctypes.cast(ctypes.byref(d_val), ctypes.c_void_p),
                        ctypes.cast(ctypes.byref(s_val), ctypes.c_void_p),
                    ]
                )

            d_out_val = ctypes.c_uint64(int(d_out))
            n_val = ctypes.c_int(total_len)
            c_vars.extend([d_out_val, n_val])
            ctypes_ptrs.extend(
                [
                    ctypes.cast(ctypes.byref(d_out_val), ctypes.c_void_p),
                    ctypes.cast(ctypes.byref(n_val), ctypes.c_void_p),
                ]
            )

            arg_array = (ctypes.c_void_p * len(ctypes_ptrs))(
                *[p.value for p in ctypes_ptrs]
            )

            threads_per_block = 256
            blocks = (total_len + threads_per_block - 1) // threads_per_block

            (launch_err,) = driver.cuLaunchKernel(
                self.kernel,
                blocks,
                1,
                1,
                threads_per_block,
                1,
                1,
                0,
                0,
                ctypes.addressof(arg_array),
                0,
            )
            if launch_err.value != 0:
                raise RuntimeError(f"CUDA kernel launch failed: {launch_err}")

            out = np.empty(total_len, dtype=np.float32)
            driver.cuMemcpyDtoH(out.ctypes.data, d_out, out_bytes)
        finally:
            for d_ptr in d_inputs:
                try:
                    driver.cuMemFree(d_ptr)
                except Exception:
                    pass
            if d_out is not None:
                try:
                    driver.cuMemFree(d_out)
                except Exception:
                    pass
            driver.cuCtxPopCurrent()

        if is_all_scalar and broadcast_shape == ():
            return float(out[0])

        return out.reshape(broadcast_shape)
